Ready-to-Run Node.js Server (server.js) 
// 1. Install dependencies: npm init -y && npm i express cors yt-dlp-exec
// 2. Run server: node server.js
const express = require('express');
const cors = require('cors');
const youtubedl = require('yt-dlp-exec');

const app = express();
app.use(cors());

app.get('/download', async (req, res) => {
  const { url, format, quality } = req.query;
  if (!url) return res.status(400).json({ error: 'Missing YouTube URL' });

  try {
    const isAudio = format === 'mp3' || format === 'm4a';
    const ext = isAudio ? format : 'mp4';
    res.header('Content-Disposition', `attachment; filename="mad-yt-${quality}.${ext}"`);

    const flags = isAudio
      ? { extractAudio: true, audioFormat: format, output: '-' }
      : { format: `bestvideo[height<=${parseInt(quality) || 1080}]+bestaudio/best`, output: '-' };

    const subprocess = youtubedl.exec(url, flags);
    subprocess.stdout.pipe(res);
  } catch (err) {
    res.status(500).json({ error: err.message });
  }
});

app.listen(4000, () => console.log('Mad YT Backend running on http://localhost:4000'));
